import { sorted } from '../../lib/glossary.mjs';

The words these docs use, in the sense Day gives them. Where a term has an everyday meaning too,
the entry says which one applies. Each entry names the page that introduces the concept; that is
the place to read next. Elsewhere in the docs, a term with a dotted underline is one of these, and
hovering it shows the definition.

<dl class="glossary">
  {sorted.map((t) => (
    <div class="glossary-entry" id={t.id}>
      <dt>
        <a href={`#${t.id}`}>{t.term}</a>
        {t.also && <span class="glossary-also">also {t.also.join(', ')}</span>}
      </dt>
      <dd>
        {t.definition.split('`').map((part, i) => (i % 2 ? <code>{part}</code> : part))}
        <span class="glossary-see">Defined in <a href={t.see.href}>{t.see.title}</a></span>
      </dd>
    </div>
  ))}
</dl>
