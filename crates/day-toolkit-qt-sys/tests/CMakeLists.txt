cmake_minimum_required(VERSION 3.16)
project(day_qt_native_tests LANGUAGES CXX)
enable_testing()
find_package(Qt6 REQUIRED COMPONENTS Widgets Svg)
add_executable(qt-resize resize.cpp)
target_compile_features(qt-resize PRIVATE cxx_std_17)
target_link_libraries(qt-resize PRIVATE Qt6::Widgets Qt6::Svg)
target_compile_options(qt-resize PRIVATE $<IF:$<CXX_COMPILER_ID:MSVC>,/UNDEBUG,-UNDEBUG>)
add_test(NAME qt-resize COMMAND qt-resize)
set_tests_properties(qt-resize PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
